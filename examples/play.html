Play

The stream stopped. It plays again here as soon as it is back.
import { WebRTCClient } from "@antmedia/webrtc-sdk"; const client = new WebRTCClient({ websocketURL }); // returned at once: listen first const session = client.play( streamId, { token, remoteVideo }); session.on("play_finished", () => console.log("source stopped, waiting")); await session.ready(); // playing const { streamInfo } = await client.getStreamInfo(streamId); await session.forceStreamQuality(720); await session.stop();
