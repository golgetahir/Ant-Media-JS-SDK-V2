import { WebRTCClient } from "@antmedia/webrtc-sdk";

const client = new WebRTCClient({ websocketURL, localVideo });

// the session exists at once: listen first
const session = client.publish("stream_7gk2p");
session.on("reconnected", () => show("live"));

await session.ready();   // live on the server
client.media.muteLocalMic();
await session.stop();
